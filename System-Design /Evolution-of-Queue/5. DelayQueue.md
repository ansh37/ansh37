# Stage 5: The Timekeeper (Priority & Delay Queue)

## Overview
Standard queues process items in the exact order they arrive (FIFO). A **Priority Queue** processes items based on an assigned rank. A **Delay Queue** is a specialized priority queue where the "rank" is a timestamp in the future, and items cannot be dequeued until that exact time is reached. 

This is the foundational architecture for scheduling systems (e.g., "Retry this payment in 5 minutes" or "Send a reminder email tomorrow").

## Key Engineering Concepts
1. **The Min-Heap:** Under the hood, priority queues are implemented as Heaps (Binary Trees in an array) to guarantee $O(\log N)$ insertions and extractions, rather than $O(N)$ sorting.
2. **`wait_until` / Time-Aware Sleeping:** When a consumer checks the queue and sees the next item is scheduled for 5 minutes from now, it must not spin-wait. It uses the OS condition variable to sleep *until* that specific timestamp.
3. **The Early Wake-up Trap:** If a consumer is sleeping for 5 minutes for Task A, and a producer suddenly pushes Task B scheduled for 1 minute from now, the producer must `notify` the sleeping consumer to wake up, re-evaluate the top of the heap, and adjust its sleep timer.

## Implementation: C++17 (Delay Queue)

```cpp
#include <queue>
#include <mutex>
#include <condition_variable>
#include <chrono>
#include <iostream>

// The task payload with an execution timestamp
struct DelayedTask {
    int id;
    std::chrono::steady_clock::time_point execute_at;

    // Min-Heap requires the greater-than operator to put smallest times at the top
    bool operator>(const DelayedTask& other) const {
        return execute_at > other.execute_at;
    }
};

class DelayQueue {
private:
    // std::priority_queue is a Max-Heap by default. We use std::greater to make it a Min-Heap.
    std::priority_queue<DelayedTask, std::vector<DelayedTask>, std::greater<DelayedTask>> pq;
    mutable std::mutex mtx;
    std::condition_variable cv;
    bool isFinished = false;

public:
    void enqueue(const DelayedTask& task) {
        std::lock_guard<std::mutex> lock(mtx);
        pq.push(task);
        // Wake up the consumer in case this new task has an earlier timestamp than the one it is sleeping for
        cv.notify_one(); 
    }

    bool dequeue(DelayedTask& out_task) {
        std::unique_lock<std::mutex> lock(mtx);

        while (true) {
            if (isFinished && pq.empty()) {
                return false;
            }

            if (pq.empty()) {
                cv.wait(lock, [this]() { return !pq.empty() || isFinished; });
                continue; // Re-evaluate loop
            }

            auto now = std::chrono::steady_clock::now();
            auto top_task = pq.top();

            if (now >= top_task.execute_at) {
                // Time has passed, safely extract
                out_task = top_task;
                pq.pop();
                return true;
            } else {
                // Task is in the future. Sleep UNTIL that exact time.
                // If a new task is pushed, cv.notify_one() will break this wait early.
                cv.wait_until(lock, top_task.execute_at, [this]() { return isFinished; });
            }
        }
    }

    void shutdown() {
        std::lock_guard<std::mutex> lock(mtx);
        isFinished = true;
        cv.notify_all();
    }
};
```
## Implementation: C++17 (Delay Queue): Optimized

```cpp
#include <queue>
#include <mutex>
#include <condition_variable>
#include <chrono>

struct DelayedTask {
    int id;
    std::chrono::steady_clock::time_point execute_at;

    bool operator>(const DelayedTask& other) const {
        return execute_at > other.execute_at;
    }
};

class OptimizedDelayQueue {
private:
    std::priority_queue<DelayedTask, std::vector<DelayedTask>, std::greater<DelayedTask>> pq;
    mutable std::mutex mtx;
    std::condition_variable cv;
    bool isFinished = false;

public:
    void enqueue(const DelayedTask& task) {
        std::lock_guard<std::mutex> lock(mtx);
        
        // Optimization: Only notify if the heap was empty OR the new task 
        // preempts the current closest execution time.
        bool notify_required = pq.empty() || (task.execute_at < pq.top().execute_at);
        
        pq.push(task);
        
        if (notify_required) {
            cv.notify_one(); // Avoids thread thrashing for downstream elements
        }
    }

    bool dequeue(DelayedTask& out_task) {
        std::unique_lock<std::mutex> lock(mtx);

        while (true) {
            if (isFinished && pq.empty()) {
                return false;
            }

            if (pq.empty()) {
                // Indefinite sleep block
                cv.wait(lock, [this]() { return !pq.empty() || isFinished; });
                continue;
            }

            auto now = std::chrono::steady_clock::now();
            DelayedTask top_task = pq.top();

            if (now >= top_task.execute_at) {
                out_task = std::move(top_task);
                pq.pop();
                return true;
            } else {
                // Timed sleep block utilizing the exact same condition variable
                // The predicate lambda handles spurious wakeups strictly during shutdown
                cv.wait_until(lock, top_task.execute_at, [this]() { return isFinished; });
            }
        }
    }
};
```

## Implementation: Python3
```py
import threading
import heapq
import time
from typing import TypeVar, Generic, Tuple, Optional, List
from dataclasses import dataclass

@dataclass
class DelayedTask:
    id: int
    execute_at_timestamp: float # Unix timestamp

    # Defines ordering for the heapq (Min-Heap)
    def __lt__(self, other):
        return self.execute_at_timestamp < other.execute_at_timestamp

class DelayQueue:
    def __init__(self):
        self._heap: List[DelayedTask] = []
        self._condition = threading.Condition()
        self._is_finished = False

    def enqueue(self, task: DelayedTask):
        with self._condition:
            heapq.heappush(self._heap, task)
            self._condition.notify()

    def dequeue(self) -> Tuple[bool, Optional[DelayedTask]]:
        with self._condition:
            while True:
                if self._is_finished and not self._heap:
                    return False, None

                if not self._heap:
                    self._condition.wait()
                    continue

                now = time.time()
                top_task = self._heap[0]

                if now >= top_task.execute_at_timestamp:
                    task = heapq.heappop(self._heap)
                    return True, task
                else:
                    # Python wait takes a timeout in seconds, not an absolute timestamp
                    sleep_duration = top_task.execute_at_timestamp - now
                    self._condition.wait(timeout=sleep_duration)
```
